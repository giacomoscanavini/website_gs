1251 · Average selling price

Match units sold to the applicable price interval, then compute revenue divided by units for every product.

Pattern: Range join / weighted mean

Example

Price 10 for 2 units and price 20 for 1 unit → 40/3 = 13.33; a product without matched sales → 0.

Study solution

SQL 
SELECT p.product_id,
       COALESCE(ROUND(SUM(p.price * u.units) / NULLIF(SUM(u.units), 0), 2), 0) AS average_price
FROM Prices p LEFT JOIN UnitsSold u
  ON p.product_id = u.product_id AND u.purchase_date BETWEEN p.start_date AND p.end_date
GROUP BY p.product_id;
pandas 
from decimal import Decimal, ROUND_HALF_UP
import pandas as pd

def round_half_up(value, digits: int = 2):
    if pd.isna(value): return float('nan')
    return float(Decimal(str(value)).quantize(Decimal(1).scaleb(-digits), rounding=ROUND_HALF_UP))

def average_selling_price(prices: pd.DataFrame, units_sold: pd.DataFrame) -> pd.DataFrame:
    p = prices.assign(start_date=pd.to_datetime(prices['start_date']), end_date=pd.to_datetime(prices['end_date']))
    u = units_sold.assign(purchase_date=pd.to_datetime(units_sold['purchase_date']))
    matched = p.merge(u, on='product_id', how='inner')
    matched = matched.loc[matched['purchase_date'].between(matched['start_date'], matched['end_date'])].copy()
    matched['paid'] = matched['price'] * matched['units']
    g = matched.groupby('product_id', as_index=False).agg(paid=('paid', 'sum'), units=('units', 'sum'))
    g['average_price'] = (g['paid'] / g['units'].replace(0, float('nan'))).map(round_half_up)
    result = p[['product_id']].drop_duplicates().merge(g[['product_id', 'average_price']], on='product_id', how='left')
    result['average_price'] = result['average_price'].fillna(0)
    return result

Why it works

Join on product, retain purchases within each inclusive date interval, aggregate revenue and units separately, and finally restore every product from Prices.

Notes

Price intervals for a product must not overlap. Products without a matching sale return zero.

    return float(Decimal(str(value)).quantize(Decimal(1).scaleb(-digits), rounding=ROUND_HALF_UP))

def average_selling_price(prices: pd.DataFrame, units_sold: pd.DataFrame) -> pd.DataFrame:
    prices = prices.assign(start_date=pd.to_datetime(prices['start_date']),
                           end_date=pd.to_datetime(prices['end_date']))
    units_sold = units_sold.assign(purchase_date=pd.to_datetime(units_sold['purchase_date']))
    df = prices.merge(units_sold, on='product_id', how='left')
    df = df[df.purchase_date.isna() | df.purchase_date.between(df.start_date, df.end_date)].copy()
    df['paid'] = df['price'] * df['units']
    totals = df.groupby('product_id', as_index=False).agg(paid=('paid', 'sum'), units=('units', 'sum'))
    totals['average_price'] = (totals['paid'] / totals['units'].replace(0, float('nan'))).map(round_half_up)
    result = prices[['product_id']].drop_duplicates().merge(totals[['product_id', 'average_price']], on='product_id', how='left')
    result['average_price'] = result['average_price'].fillna(0)

import pandas as pd

def average_selling_price(prices: pd.DataFrame, units_sold: pd.DataFrame) -> pd.DataFrame:
    df = prices.merge(units_sold, on='product_id', how='left')
    df = df[df.purchase_date.isna() | ((df.end_date >= df.purchase_date) & (df.purchase_date >= df.start_date))]
    df['paid'] = df['price'] * df['units']
    
    return df.groupby(by='product_id').apply(
        lambda x: round(x['paid'].sum() / x['units'].sum(), 2) if x['units'].sum() != 0 else 0
    ).reset_index(name='average_price')
