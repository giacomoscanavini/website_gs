1341 · Movie rating report

Return the most prolific reviewer overall, then the movie with highest average rating in February 2020.

Pattern: Two aggregations / tie-breaking

Example

Alice and Bob tie in review count → Alice. Two February movies tie in average rating → alphabetically smaller title.

Study solution

SQL 
WITH user_winner AS (
    SELECT u.name AS results FROM MovieRating r JOIN Users u ON r.user_id = u.user_id
    GROUP BY u.user_id, u.name ORDER BY COUNT(*) DESC, u.name LIMIT 1
), movie_winner AS (
    SELECT m.title AS results FROM MovieRating r JOIN Movies m ON r.movie_id = m.movie_id
    WHERE r.created_at >= '2020-02-01' AND r.created_at < '2020-03-01'
    GROUP BY m.movie_id, m.title ORDER BY AVG(r.rating) DESC, m.title LIMIT 1
), combined AS (
    SELECT 1 AS sequence_no, results FROM user_winner
    UNION ALL SELECT 2, results FROM movie_winner
)
SELECT results FROM combined ORDER BY sequence_no;
pandas 
import pandas as pd

def movie_rating(movies: pd.DataFrame, users: pd.DataFrame, movie_rating: pd.DataFrame) -> pd.DataFrame:
    r = movie_rating.assign(created_at=pd.to_datetime(movie_rating['created_at']))
    users_ranked = r.groupby('user_id').size().reset_index(name='n_reviews').merge(users, on='user_id')
    first = users_ranked.sort_values(['n_reviews', 'name'], ascending=[False, True]).head(1)[['name']].rename(columns={'name': 'results'})
    feb = r.loc[r['created_at'].ge('2020-02-01') & r['created_at'].lt('2020-03-01')]
    movies_ranked = feb.groupby('movie_id', as_index=False).agg(avg_rating=('rating', 'mean')).merge(movies, on='movie_id')
    second = movies_ranked.sort_values(['avg_rating', 'title'], ascending=[False, True]).head(1)[['title']].rename(columns={'title': 'results'})
    return pd.concat([first, second], ignore_index=True)

Why it works

Solve the two leaderboards independently, applying their different date scopes. Use ascending name/title to break ties and concatenate without deduplication.

Notes

User review counts use all dates. Movie averages use February 2020 only. UNION ALL preserves both result rows.

def movie_rating(movies: pd.DataFrame, users: pd.DataFrame, movie_rating: pd.DataFrame) -> pd.DataFrame:
    movie_rating = movie_rating.assign(created_at=pd.to_datetime(movie_rating['created_at']))

        nReviews = ('rating', 'size')

    df_movies =   movie_rating[(movie_rating.created_at >=   '2020-02-01') & (movie_rating.created_at < '2020-03-01')]
    df_movies = df_movies.groupby(by='movie_id').agg(
        avgRating = ('rating', 'mean')
    )
    df_movies =   df_movies.merge(movies, on='movie_id', how='left').sort_values(by=['avgRating', 'title'], ascending=[False, True])[['title', 'avgRating']]
    return   pd.concat([df_users.head(1)[['name']].rename(columns={'name': 'results'}),
                        df_movies.head(1)[['title']].rename(columns={'title': 'results'}),
                    ])

import pandas as pd

def movie_rating(movies: pd.DataFrame, users: pd.DataFrame, movie_rating: pd.DataFrame) -> pd.DataFrame:
    df_users = movie_rating.groupby(by='user_id').agg(
        nReviews = ('rating', 'size')
    ).reset_index()
    df_users = df_users.merge(users, on='user_id', how='left').sort_values(by=['nReviews', 'name'], ascending=[False, True])[['name', 'nReviews']]

    df_movies = movie_rating[(movie_rating.created_at >= '2020-02-01') & (movie_rating.created_at < '2020-03-01')]
    df_movies = df_movies.groupby(by='movie_id').agg(
        avgRating = ('rating', 'mean')
    )
    df_movies = df_movies.merge(movies, on='movie_id', how='left').sort_values(by=['avgRating', 'title'], ascending=[False, True])[['title', 'avgRating']]

    return pd.concat([df_users.iloc[[0]][['name']].rename(columns={'name': 'results'}),
