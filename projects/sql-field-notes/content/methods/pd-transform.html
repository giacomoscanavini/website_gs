def sales_analysis(sales: pd.DataFrame) -> pd.DataFrame:
    first_year = sales.groupby('product_id')['year'].transform('min')
    return sales.loc[sales['year'].eq(first_year), ['product_id', 'year', 'quantity', 'price']].rename(columns={'year': 'first_year'})

def find_primary_department(employee: pd.DataFrame) -> pd.DataFrame:
    n = employee.groupby('employee_id')['department_id'].transform('size')
    mask = employee['primary_flag'].eq('Y') | n.eq(1)
    return employee.loc[mask, ['employee_id', 'department_id']].copy()
